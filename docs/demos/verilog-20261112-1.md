```python
import cocotb
from cocotb.clock import Clock
from cocotb.triggers import ClockCycles


async def start(dut):
    cocotb.start_soon(Clock(dut.clk, 100, unit="ns").start())
    dut.ena.value = 1
    dut.ui_in.value = 0
    dut.uio_in.value = 0

    dut.rst_n.value = 1
    await ClockCycles(dut.clk, 1)


@cocotb.test()
async def test_reset(dut):
    """Covers: R1"""
    await start(dut)
    assert int(dut.uo_out.value) == 0
    dut.ui_in.value = 1
    await ClockCycles(dut.clk, 5)
    dut.rst_n.value = 0
    await ClockCycles(dut.clk, 2)
    assert int(dut.uo_out.value) == 0


@cocotb.test()
async def test_count_and_hold(dut):
    """Covers: R2, R3"""
    await start(dut)
    dut.ui_in.value = 1
    await ClockCycles(dut.clk, 5)
    v = int(dut.uo_out.value)
    assert v == 5, f"expected 5 got {v}"
    dut.ui_in.value = 0
    await ClockCycles(dut.clk, 1)
    v = int(dut.uo_out.value)
    await ClockCycles(dut.clk, 10)
    assert int(dut.uo_out.value) == v
    dut.ui_in.value = 1
    await ClockCycles(dut.clk, 1)
    assert int(dut.uo_out.value) == (v + 1) & 0xFF
```

```verilog
// SPDX-FileCopyrightText: Adam Engelbrecht
// SPDX-License-Identifier: Apache-2.0

// counter8: 8-bit counter on uo_out, counts while ui_in[0] is high.

`default_nettype none

module counter8 (
    input  wire [7:0] ui_in,    // Dedicated inputs
    output wire [7:0] uo_out,   // Dedicated outputs
    input  wire [7:0] uio_in,   // IOs: Input path
    output wire [7:0] uio_out,  // IOs: Output path
    output wire [7:0] uio_oe,   // IOs: Enable path (active high: 0=input, 1=output)
    input  wire       ena,      // always 1 when the design is powered
    input  wire       clk,      // clock
    input  wire       rst_n     // reset_n - low to reset
);

  reg [7:0] count;

  always @(posedge clk) begin
    if (!rst_n) count <= 8'd0;
    else if (ui_in[0]) count <= count + 8'd1;
  end

  assign uo_out  = count;
  assign uio_out = 8'd0;
  assign uio_oe  = 8'd0;

  wire _unused = &{ena, ui_in[7:1], uio_in, 1'b0};

endmodule
```
