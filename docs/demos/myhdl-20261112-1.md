import cocotb
from cocotb.clock import Clock
from cocotb.triggers import ClockCycles, FallingEdge, ReadOnly, RisingEdge


async def reset(dut):
    cocotb.start_soon(Clock(dut.clk, 10, unit="ns").start())
    dut.ena.value = 1
    dut.ui_in.value = 0
    dut.uio_in.value = 0
    dut.rst_n.value = 0
    await ClockCycles(dut.clk, 5)
    await FallingEdge(dut.clk)
    dut.rst_n.value = 1


async def sample(dut):
    await RisingEdge(dut.clk)
    await ReadOnly()
    return int(dut.uo_out.value)


@cocotb.test()
async def test_reset_value(dut):
    """Covers: R1, R5"""
    await reset(dut)
    actual = await sample(dut)
    assert actual == 0, "uo_out after reset with enable low"


@cocotb.test()
async def test_counts_when_enabled(dut):
    """Covers: R1, R2"""
    await reset(dut)
    await FallingEdge(dut.clk)
    dut.ui_in.value = 1
    seen = [await sample(dut) for _ in range(10)]
    assert seen == list(range(1, 11)), "uo_out on 10 enabled clocks"


@cocotb.test()
async def test_holds_when_disabled(dut):
    """Covers: R3"""
    await reset(dut)
    await FallingEdge(dut.clk)
    dut.ui_in.value = 1
    for _ in range(7):
        await sample(dut)
    await FallingEdge(dut.clk)
    dut.ui_in.value = 0
    expected = await sample(dut)
    assert expected == 7, "count after 7 enabled clocks"
    await ClockCycles(dut.clk, 20)
    await ReadOnly()
    actual = int(dut.uo_out.value)
    assert actual == expected, "uo_out after 20 clocks with enable low"


@cocotb.test()

from myhdl import Signal, always_comb, always_seq, block, modbv


@block
def counter8(clk, rst_n, ui_in, uo_out):
    count = Signal(modbv(0)[8:])

    @always_seq(clk.posedge, reset=rst_n)
    def seq():
        if ui_in[0]:
            count.next = count + 1

    @always_comb
    def drive():
        uo_out.next = count

    return seq, drive
```
