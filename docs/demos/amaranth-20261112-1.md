```python
import cocotb
from cocotb.clock import Clock
from cocotb.triggers import ClockCycles, FallingEdge, ReadOnly, RisingEdge


async def reset(dut):
    """Start a 10 ns clock and hold the active-low reset for 5 cycles."""
    cocotb.start_soon(Clock(dut.clk, 10, unit="ns").start())
    dut.ena.value = 1
    dut.ui_in.value = 0
    dut.uio_in.value = 0
    dut.rst_n.value = 0
    await ClockCycles(dut.clk, 5)
    await FallingEdge(dut.clk)
    dut.rst_n.value = 1


async def sample(dut):
    """Wait for a rising edge, let outputs settle, read uo_out."""
    await RisingEdge(dut.clk)
    await ReadOnly()
    return int(dut.uo_out.value)


@cocotb.test()
async def test_counts_when_enabled(dut):
    """Covers: R1"""
    await reset(dut)
    start = await sample(dut)
    await FallingEdge(dut.clk)
    dut.ui_in.value = 1
    for i in range(1, 11):
        actual = await sample(dut)
        assert actual == (start + i) % 256, f"uo_out after {i} enabled clocks"


@cocotb.test()
async def test_holds_when_disabled(dut):
    """Covers: R2"""
    await reset(dut)
    await FallingEdge(dut.clk)
    dut.ui_in.value = 1
    for _ in range(7):
        await sample(dut)
    await FallingEdge(dut.clk)
    dut.ui_in.value = 0
    expected = await sample(dut)
    for _ in range(20):
        actual = await sample(dut)
        assert actual == expected, "uo_out while count disabled"
    # only bit 0 of ui_in is the enable: other bits must not count
    await FallingEdge(dut.clk)
    dut.ui_in.value = 0xFE
    for _ in range(5):
        actual = await sample(dut)
        assert actual == expected, "uo_out with ui_in[7:1] high but ui_in[0] low"


@cocotb.test()
async def test_wraps_to_zero(dut):
    """Covers: R3"""
    await reset(dut)
    await FallingEdge(dut.clk)
    dut.ui_in.value = 1
    for _ in range(300):
        if await sample(dut) == 255:
            break
    else:
        raise AssertionError("the count never reached 255")
    actual = await sample(dut)
    assert actual == 0, "uo_out one clock after 255"

```python
from amaranth.hdl import ClockDomain, Module, Signal
from amaranth.lib import wiring
from amaranth.lib.wiring import In, Out


class Core(wiring.Component):
    clk: In(1)
    rst_n: In(1)
    ena: In(1)
    ui_in: In(8)
    uio_in: In(8)
    uo_out: Out(8)
    uio_out: Out(8)
    uio_oe: Out(8)

    def elaborate(self, platform):
        m = Module()
        m.domains.sync = cd = ClockDomain("sync")
        m.d.comb += [cd.clk.eq(self.clk), cd.rst.eq(~self.rst_n)]

        count = Signal(8)
        with m.If(self.ui_in[0]):
            m.d.sync += count.eq(count + 1)

        m.d.comb += [
            self.uo_out.eq(count),
            self.uio_out.eq(0),
            self.uio_oe.eq(0),
        ]
        return m


def top() -> wiring.Component:
    return Core()
